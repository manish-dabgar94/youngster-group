Spring Boot 3 + PostgreSQL Architecture

Enterprise Backend Design & REST API DDL for Youngster Group

1. Relational Database Schema (PostgreSQL DDL with Global Party Expense) 

-- Youngster Group Database Schema
CREATE TABLE members (
    id SERIAL PRIMARY KEY,
    gujarati_name VARCHAR(120) NOT NULL,
    phone_number VARCHAR(15),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE monthly_ledgers (
    id SERIAL PRIMARY KEY,
    cycle_date DATE NOT NULL UNIQUE,       -- e.g., '2026-09-01'
    previous_month_cash NUMERIC(12, 2) DEFAULT 220.00,
    party_kharch NUMERIC(10, 2) DEFAULT 0.00,  -- Dedicated Overall Party / Event Expense
    post_office_balance NUMERIC(12, 2) DEFAULT 310000.00,
    cash_in_hand NUMERIC(12, 2) DEFAULT 30700.00,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE member_contributions (
    id SERIAL PRIMARY KEY,
    ledger_id INT REFERENCES monthly_ledgers(id) ON DELETE CASCADE,
    member_id INT REFERENCES members(id) ON DELETE RESTRICT,
    amount NUMERIC(10, 2) NOT NULL DEFAULT 1000.00,
    is_paid BOOLEAN DEFAULT TRUE,
    paid_date TIMESTAMP WITH TIME ZONE,
    CONSTRAINT unique_member_cycle UNIQUE (ledger_id, member_id)
);

CREATE TABLE loan_distributions (
    id SERIAL PRIMARY KEY,
    ledger_id INT REFERENCES monthly_ledgers(id) ON DELETE CASCADE,
    member_id INT REFERENCES members(id) ON DELETE RESTRICT,
    principal_amount NUMERIC(12, 2) NOT NULL,
    interest_rate NUMERIC(4, 2) DEFAULT 2.00, -- 2% monthly interest
    repaid_jama NUMERIC(12, 2) DEFAULT 0.00,
    status VARCHAR(20) DEFAULT 'ACTIVE'
);

2. Spring Boot 3 Entity & Repository Sample 

@Entity
@Table(name = "monthly_ledgers")
@Getter @Setter @NoArgsConstructor
public class MonthlyLedger {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    private LocalDate cycleDate;
    private BigDecimal previousMonthCash;
    private BigDecimal partyKharch = BigDecimal.ZERO;
    private BigDecimal postOfficeBalance;
    private BigDecimal cashInHand;

    @OneToMany(mappedBy = "ledger", cascade = CascadeType.ALL)
    private List<LoanDistribution> loans;
}

3. Recommended RESTful APIs

Method Path Description
GET /api/v1/ledger/{monthYear} Get complete report snapshot (Savings, Loans, Totals)
PUT /api/v1/contributions/{id}/toggle Toggle payment status & recalculate balances
POST /api/v1/loans Distribute new member loan or modify amount
